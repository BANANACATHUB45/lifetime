--// Button hover effect
local function hoverButton(button, normalColor, hoverColor)
    button.MouseEnter:Connect(function()
        TweenService:Create(
            button,
            TweenInfo.new(0.15),
            {BackgroundColor3 = hoverColor}
        ):Play()
    end)

    button.MouseLeave:Connect(function()
        TweenService:Create(
            button,
            TweenInfo.new(0.15),
            {BackgroundColor3 = normalColor}
        ):Play()
    end)
end

hoverButton(
    CheckButton,
    YELLOW,
    Color3.fromRGB(255, 225, 75)
)

    CopyButton,
    Color3.fromRGB(40, 40, 40),
    Color3.fromRGB(55, 55, 55)
)

--// Check key
CheckButton.MouseButton1Click:Connect(function()

    local enteredKey = KeyBox.Text

    if enteredKey == "" then
        Status.Text = "⚠ Please enter your key."
        Status.TextColor3 = RED
        return
    end

    if enteredKey == CORRECT_KEY then

        Status.Text = "✓ Key verified! Access granted."
        Status.TextColor3 = GREEN

        CheckButton.Text = "✓  VERIFIED"

        task.wait(1)

        -- Put your main script here
        print("Banana Cat Hub: Key accepted!")

    else

        Status.Text = "✕ Invalid key. Please try again."
        Status.TextColor3 = RED

        -- Small shake animation
        local originalPosition = Main.Position

        for i = 1, 3 do
            Main.Position = originalPosition + UDim2.fromOffset(5, 0)
            task.wait(0.04)
            Main.Position = originalPosition - UDim2.fromOffset(5, 0)
            task.wait(0.04)
        end

        Main.Position = originalPosition
    end
end)

--// Make GUI draggable
local dragging = false
local dragStart
local startPosition

TopBar.InputBegan:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then

        dragging = true
        dragStart = input.Position
        startPosition = Main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TopBar.InputChanged:Connect(function(input)

    if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then

        dragStart = dragStart or input.Position

        input.Changed:Connect(function()
            if dragging then
                local delta = input.Position - dragStart

                Main.Position = UDim2.new(
                    startPosition.X.Scale,
                    startPosition.X.Offset + delta.X,
                    startPosition.Y.Scale,
                    startPosition.Y.Offset + delta.Y
                )
            end
        end)
    end
end)
